commands:
--tokenize auth
curl -k http://localhost:8088/services/collector -H "Authorization:Splunk <token>" 
-d "{\"sourcetype\": \"trial\",\"event\":\"hello world\"}"

For example:
adminu@n1:~$ curl -k http://n1:8088/services/collector -H "Authorization:Splunk fce9df17-751a-49ab-9466-07264489f888" 
-d "{\"sourcetype\": \"trial\",\"event\":\"hello world\"}"

{"text":"Success","code":0}

--basic auth
curl -k http://localhost:8088/services/collector -u "admin:<token>" 
-d "{\"sourcetype\": \"trial\",\"event\":\"hello world\"}"

For example:
adminu@n1:~$ curl -k http://n1:8088/services/collector -u "admin:fce9df17-751a-49ab-9466-07264489f888" 
-d "{\"sourcetype\": \"trial\",\"event\":\"new hello world\"}"

{"text":"Success","code":0} 

--multiple events
curl -k http://localhost:8088/services/collector -u "admin<token>" -d 
"{\"sourcetype\": \"trial\",\"event\":\"hello world3\"}"
"{\"sourcetype\": \"trial\",\"event\":\"hello world4\"}"

For example:
adminu@n1:~$ curl -k http://n1:8088/services/collector -u "admin:fce9df17-751a-49ab-9466-07264489f888" 
-d "{\"sourcetype\": \"trial\",\"event\":\"new hello world3\"}"
"{\"sourcetype\": \"trial\",\"event\":\"new hello world4\"}"

--raw text
curl -k "http://localhost:8088/services/collector/raw?sourcetype=trial" -H "Authorization:Splunk <token>" 
-d "1,2,3... Hello World"

For example:
adminu@n1:~$ curl -k "http://n1:8088/services/collector/raw?sourcetype=trial" 
-H "Authorization:Splunk fce9df17-751a-49ab-9466-07264489f888" -d "1,2,3,...Hello World!"

{"text":"Success","code":0}
